	\section*{Quick Reference: Exponent Rules}

		\subsection*{The Laws of Exponents}

			\textbf{Product Rule:}
            
            $x^a \cdot x^b = x^{a+b}$\\

			\textbf{Quotient Rule:}
            
            $\frac{x^a}{x^b} = x^{a-b}$ (when $x \neq 0$)\\

			\textbf{Power Rule:}
            
            $(x^a)^b = x^{ab}$\\

			\textbf{Power of a Product:}
            
            $(xy)^a = x^a y^a$\\

			\textbf{Negative Exponent:}
            
            $x^{-a} = \frac{1}{x^a}$ (when $x \neq 0$)\\

		\subsection*{Polynomial Operations}

			\textbf{Multiplying:} Use distributive property

		\subsection*{Greatest Common Factor (GCF)}

			To factor using GCF:
			\begin{enumerate}
				\item Find the largest number that divides all coefficients
				\item Find the lowest power of each variable in all terms
				\item Factor out the GCF
			\end{enumerate}

	\section*{Session 9.2: \\ Factoring Trinomials \& Special Products}

	\section*{Quick Reference: Factoring Trinomials}

		\subsection*{Factoring $x^2 + bx + c$}

			To factor a trinomial of the form $x^2 + bx + c$:

			\textbf{Steps:}
			\begin{enumerate}
				\item Find two numbers that multiply to give $c$
				\item AND add to give $b$
				\item Write as $(x + \text{first number})(x + \text{second number})$
			\end{enumerate}

		\subsection*{Factoring $ax^2 + bx + c$ (when $a \neq 1$)}

			\textbf{AC Method:}
			\begin{enumerate}
				\item Multiply $a \times c$
				\item Find two numbers that multiply to $ac$ and add to $b$
				\item Rewrite the middle term using these two numbers
				\item Factor by grouping
			\end{enumerate}

		\subsection*{Special Products (Shortcuts)}

			\textbf{Perfect Square Trinomials:}

			$a^2 + 2ab + b^2 = (a + b)^2$

			$a^2 - 2ab + b^2 = (a - b)^2$

			\textbf{Difference of Squares:}

			$a^2 - b^2 = (a + b)(a - b)$

		\subsection*{Zero Product Property}

			If $a \cdot b = 0$, then $a = 0$ or $b = 0$ (or both).

			This is the key to solving quadratic equations by factoring!

		\subsection*{Steps to Solve by Factoring}

			\begin{enumerate}
				\item Write the equation in standard form: $ax^2 + bx + c = 0$
				\item Factor the left side completely
				\item Set each factor equal to zero
				\item Solve each equation
				\item Check your solutions (optional but recommended)
			\end{enumerate}

	\section*{Homework 9.1 Answer Key: \\ Exponents, Polynomial Operations, \& GCF Factoring}

	\subsection*{Part A: Simplify Using Exponent Rules}

		\textbf{Problem 1:} $x^4 \cdot x^6$
		
		\textbf{Solution:} Using the product rule, add the exponents.
		
		$x^4 \cdot x^6 = x^{4+6} = x^{10}$
		
		\textbf{Answer:} $x^{10}$

		\vspace{0.5cm}

		\textbf{Problem 2:} $\frac{y^9}{y^5}$
		
		\textbf{Solution:} Using the quotient rule, subtract the exponents.
		
		$\frac{y^9}{y^5} = y^{9-5} = y^4$
		
		\textbf{Answer:} $y^4$

		\vspace{0.5cm}

		\textbf{Problem 3:} $(m^3)^4$
		
		\textbf{Solution:} Using the power rule, multiply the exponents.
		
		$(m^3)^4 = m^{3 \cdot 4} = m^{12}$
		
		\textbf{Answer:} $m^{12}$

		\vspace{0.5cm}

		\textbf{Problem 4:} $(2x^2y^3)^3$
		
		\textbf{Solution:} Apply the power to each factor.
		
		$(2x^2y^3)^3 = 2^3 \cdot (x^2)^3 \cdot (y^3)^3 = 8x^6y^9$
		
		\textbf{Answer:} $8x^6y^9$

		\vspace{0.5cm}

		\textbf{Problem 5:} $\frac{12x^7}{3x^2}$
		
		\textbf{Solution:} Divide coefficients and subtract exponents.
		
		$\frac{12x^7}{3x^2} = \frac{12}{3} \cdot x^{7-2} = 4x^5$
		
		\textbf{Answer:} $4x^5$

	\newpage

	\subsection*{Part B: Polynomial Operations}

		\textbf{Problem 6:} $3x(2x^2 - 5x + 4)$
		
		\textbf{Solution:} Distribute $3x$ to each term.
		
		$3x(2x^2) + 3x(-5x) + 3x(4) = 6x^3 - 15x^2 + 12x$
		
		\textbf{Answer:} $6x^3 - 15x^2 + 12x$

		\vspace{0.5cm}

		\textbf{Problem 7:} $(x + 4)(x + 7)$
		
		\textbf{Solution:} Use FOIL method.
		
		First: $x \cdot x = x^2$
		
		Outer: $x \cdot 7 = 7x$
		
		Inner: $4 \cdot x = 4x$
		
		Last: $4 \cdot 7 = 28$
		
		Combine: $x^2 + 7x + 4x + 28 = x^2 + 11x + 28$
		
		\textbf{Answer:} $x^2 + 11x + 28$

		\vspace{0.5cm}

		\textbf{Problem 8:} $(2x - 3)(x + 5)$
		
		\textbf{Solution:} Use FOIL method.
		
		First: $2x \cdot x = 2x^2$
		
		Outer: $2x \cdot 5 = 10x$
		
		Inner: $-3 \cdot x = -3x$
		
		Last: $-3 \cdot 5 = -15$
		
		Combine: $2x^2 + 10x - 3x - 15 = 2x^2 + 7x - 15$
		
		\textbf{Answer:} $2x^2 + 7x - 15$

	\newpage

	\subsection*{Part C: GCF Factoring}

		\textbf{Problem 9:} $15y^2 - 10y$
		
		\textbf{Solution:} GCF is $5y$.
		
		$15y^2 - 10y = 5y(3y) + 5y(-2) = 5y(3y - 2)$
		
		\textbf{Answer:} $5y(3y - 2)$

		\vspace{0.5cm}

		\textbf{Problem 10:} $8x^3 + 12x^2 - 4x$
		
		\textbf{Solution:} GCF is $4x$.
		
		$8x^3 + 12x^2 - 4x = 4x(2x^2) + 4x(3x) + 4x(-1) = 4x(2x^2 + 3x - 1)$
		
		\textbf{Answer:} $4x(2x^2 + 3x - 1)$

		\vspace{0.5cm}

		\textbf{Problem 11:} $14x^2y + 21xy^2$
		
		\textbf{Solution:} GCF is $7xy$.
		
		$14x^2y + 21xy^2 = 7xy(2x) + 7xy(3y) = 7xy(2x + 3y)$
		
		\textbf{Answer:} $7xy(2x + 3y)$

	\newpage

	\subsection*{Part D: Word Problem}

		\textbf{Problem 12:}
		
		The area of a square is given by $6x^2 + 9x$ square inches.
		
		\textbf{Part A:} Factor this expression.
		
		\textbf{Solution:} GCF is $3x$.
		
		$6x^2 + 9x = 3x(2x + 3)$
		
		\textbf{Answer:} $3x(2x + 3)$

		\vspace{0.5cm}

		\textbf{Part B:} Find one possible combination for the length, width, and area of the square.
		
		\textbf{Solution:} Note: The problem statement says "square" but the factored form gives different dimensions, suggesting this is actually a rectangle.
		
		From the factored form $3x(2x + 3)$:
		
		Width = $3x$ inches
		
		Length = $(2x + 3)$ inches
		
		Area = $6x^2 + 9x$ square inches
		
		For a specific example, if $x = 1$:
		
		Width = $3(1) = 3$ inches
		
		Length = $2(1) + 3 = 5$ inches
		
		Area = $6(1)^2 + 9(1) = 15$ square inches
		
		\textbf{Answer:} Width = $3x$, Length = $(2x + 3)$; or if $x = 1$: Width = 3 in, Length = 5 in, Area = 15 sq in

\section*{Homework 9.2 Answer Key: \\ Factoring Trinomials \& Special Products}

	\subsection*{Part A: Factor Simple Trinomials ($x^2 + bx + c$)}

		\textbf{Problem 1:} $x^2 + 8x + 15$
		
		\textbf{Solution:} Find two numbers that multiply to 15 and add to 8.
		
		Numbers: 3 and 5 (since $3 \times 5 = 15$ and $3 + 5 = 8$)
		
		\textbf{Answer:} $(x + 3)(x + 5)$

		\vspace{0.5cm}

		\textbf{Problem 2:} $x^2 - 7x + 12$
		
		\textbf{Solution:} Find two numbers that multiply to 12 and add to $-7$.
		
		Numbers: $-3$ and $-4$ (since $(-3) \times (-4) = 12$ and $-3 + (-4) = -7$)
		
		\textbf{Answer:} $(x - 3)(x - 4)$

		\vspace{0.5cm}

		\textbf{Problem 3:} $x^2 + 2x - 15$
		
		\textbf{Solution:} Find two numbers that multiply to $-15$ and add to 2.
		
		Numbers: 5 and $-3$ (since $5 \times (-3) = -15$ and $5 + (-3) = 2$)
		
		\textbf{Answer:} $(x + 5)(x - 3)$

	\newpage

	\subsection*{Part B: Factor When $a \neq 1$}

		\textbf{Problem 4:} $2x^2 + 7x + 3$
		
		\textbf{Solution:} Use AC method. $a \times c = 2 \times 3 = 6$
		
		Find two numbers that multiply to 6 and add to 7: 6 and 1
		
		$2x^2 + 6x + 1x + 3$
		
		$2x(x + 3) + 1(x + 3)$
		
		\textbf{Answer:} $(2x + 1)(x + 3)$

		\vspace{0.5cm}

		\textbf{Problem 5:} $3x^2 + 10x + 8$
		
		\textbf{Solution:} Use AC method. $a \times c = 3 \times 8 = 24$
		
		Find two numbers that multiply to 24 and add to 10: 6 and 4
		
		$3x^2 + 6x + 4x + 8$
		
		$3x(x + 2) + 4(x + 2)$
		
		\textbf{Answer:} $(3x + 4)(x + 2)$

		\vspace{0.5cm}

		\textbf{Problem 6:} $2x^2 - 5x - 3$
		
		\textbf{Solution:} Use AC method. $a \times c = 2 \times (-3) = -6$
		
		Find two numbers that multiply to $-6$ and add to $-5$: $-6$ and 1
		
		$2x^2 - 6x + 1x - 3$
		
		$2x(x - 3) + 1(x - 3)$
		
		\textbf{Answer:} $(2x + 1)(x - 3)$

		\textbf{Solution:}
		
		Factor: $(x + 2)(x + 5) = 0$
		
		Set each factor equal to zero:
		
		$x + 2 = 0$ or $x + 5 = 0$
		
		$x = -2$ or $x = -5$
		
		\textbf{Answer:} $x = -2$ or $x = -5$

		\vspace{0.5cm}

		\textbf{Problem 2:} $x^2 - 4x - 12 = 0$
		
		\textbf{Solution:}
		
		Factor: $(x - 6)(x + 2) = 0$
		
		Set each factor equal to zero:
		
		$x - 6 = 0$ or $x + 2 = 0$
		
		$x = 6$ or $x = -2$
		
		\textbf{Answer:} $x = 6$ or $x = -2$

		\vspace{0.5cm}

		\textbf{Problem 3:} $x^2 - 64 = 0$
		
		\textbf{Solution:}
		
		Factor (difference of squares): $(x + 8)(x - 8) = 0$
		
		Set each factor equal to zero:
		
		$x + 8 = 0$ or $x - 8 = 0$
		
		$x = -8$ or $x = 8$
		
		\textbf{Answer:} $x = -8$ or $x = 8$

		\vspace{0.5cm}

		\textbf{Problem 4:} $2x^2 + 9x + 4 = 0$
		
		\textbf{Solution:}
		
		Factor: $(2x + 1)(x + 4) = 0$
		
		Set each factor equal to zero:
		
		$2x + 1 = 0$ or $x + 4 = 0$

		\textbf{Problem 5:} $x^2 + 8x = -15$
		
		\textbf{Solution:}
		
		Rewrite in standard form: $x^2 + 8x + 15 = 0$
		
		Factor: $(x + 3)(x + 5) = 0$
		
		Set each factor equal to zero:
		
		$x + 3 = 0$ or $x + 5 = 0$
		
		$x = -3$ or $x = -5$
		
		\textbf{Answer:} $x = -3$ or $x = -5$

		\vspace{0.5cm}

		\textbf{Problem 6:} $x^2 = 5x + 14$
		
		\textbf{Solution:}
		
		Rewrite in standard form: $x^2 - 5x - 14 = 0$
		
		Factor: $(x - 7)(x + 2) = 0$
		
		Set each factor equal to zero:
		
		$x - 7 = 0$ or $x + 2 = 0$
		
		$x = 7$ or $x = -2$
		
		\textbf{Answer:} $x = 7$ or $x = -2$

	\subsection*{Part C: Equations with GCF}

		\textbf{Problem 7:} $3x^2 = 27x$
		
		\textbf{Solution:}
		
		Rewrite in standard form: $3x^2 - 27x = 0$
		
		Factor out GCF: $3x(x - 9) = 0$
		
		Set each factor equal to zero:
		
		$3x = 0$ or $x - 9 = 0$
		
		$x = 0$ or $x = 9$
		
		\textbf{Answer:} $x = 0$ or $x = 9$

	\subsection*{Part D: Application Problems}

		\textbf{Problem 8: Area of a Rectangle}
		
		A rectangle has a length that is 4 meters more than its width.
		
		The area of the rectangle is 60 meters squared.
		
		Find the dimensions of the rectangle.
		
		\textbf{Solution:}
		
		Let $w = $ width in meters
		
		Then length $= w + 4$ meters
		
		Area equation: $w(w + 4) = 60$
		
		Rewrite in standard form: $w^2 + 4w - 60 = 0$
		
		Factor: $(w + 10)(w - 6) = 0$
		
		Set each factor equal to zero:
		
		$w + 10 = 0$ or $w - 6 = 0$
		
		$w = -10$ or $w = 6$
		
		Which solution makes sense in context? $w = 6$ (width cannot be negative)
		
		Width: 6 meters
		
		Length: $6 + 4 = 10$ meters
		
		\textbf{Answer:} Width = 6 meters, Length = 10 meters

		\vspace{0.5cm}

		\textbf{Problem 9: Projectile Motion}
		
		A ball is thrown upward from the ground.
		
		Its height $h$ in feet after $t$ seconds is given by $h = -16t^2 + 48t$.
		
		When will the ball hit the ground?
		
		\textbf{Solution:}
		
		The ball hits the ground when $h = 0$:
		
		$-16t^2 + 48t = 0$
		
		Factor out GCF: $-16t(t - 3) = 0$
		
		Set each factor equal to zero:
		
		$-16t = 0$ or $t - 3 = 0$
		
		$t = 0$ or $t = 3$
		
		$t = 0$ is when the ball is thrown (starting position)
		
		$t = 3$ is when the ball hits the ground
		
		\textbf{Answer:} The ball hits the ground at $t = 3$ seconds

\section*{Unit 9 Quiz Answer Key: Quadratics \& Polynomials}

	\subsection*{Section 1: Exponent Rules}

		\textbf{Problem 1:} $x^5 \cdot x^8$
		
		\textbf{Solution:} Using the product rule, add the exponents.
		
		$x^5 \cdot x^8 = x^{5+8} = x^{13}$
		
		\textbf{Answer:} $x^{13}$

		\vspace{0.5cm}

		\textbf{Problem 2:} $(3xy^2)^3$
		
		\textbf{Solution:} Apply the power to each factor.
		
		$(3xy^2)^3 = 3^3 \cdot x^3 \cdot (y^2)^3 = 27x^3y^6$
		
		\textbf{Answer:} $27x^3y^6$

		\vspace{0.5cm}

		\textbf{Problem 3:} $\frac{20m^9}{4m^3}$
		
		\textbf{Solution:} Divide coefficients and subtract exponents.
		
		$\frac{20m^9}{4m^3} = \frac{20}{4} \cdot m^{9-3} = 5m^6$
		
		\textbf{Answer:} $5m^6$

	\newpage

	\subsection*{Section 2: Polynomial Operations}

		\textbf{Problem 4:} $5x(3x^2 - 2x + 7)$
		
		\textbf{Solution:} Distribute $5x$ to each term.
		
		$5x(3x^2) + 5x(-2x) + 5x(7) = 15x^3 - 10x^2 + 35x$
		
		\textbf{Answer:} $15x^3 - 10x^2 + 35x$

		\vspace{0.5cm}

		\textbf{Problem 5:} $(3x - 2)(2x + 5)$
		
		\textbf{Solution:} Use FOIL method.
		
		First: $3x \cdot 2x = 6x^2$
		
		Outer: $3x \cdot 5 = 15x$
		
		Inner: $-2 \cdot 2x = -4x$
		
		Last: $-2 \cdot 5 = -10$
		
		Combine: $6x^2 + 15x - 4x - 10 = 6x^2 + 11x - 10$
		
		\textbf{Answer:} $6x^2 + 11x - 10$

	\newpage

	\subsection*{Section 3: GCF Factoring}

		\textbf{Problem 6:} $18x^3 - 12x^2$
		
		\textbf{Solution:} GCF is $6x^2$.
		
		$18x^3 - 12x^2 = 6x^2(3x) + 6x^2(-2) = 6x^2(3x - 2)$
		
		\textbf{Answer:} $6x^2(3x - 2)$

		\vspace{0.5cm}

		\textbf{Problem 7:} $15a^2b^3 + 25ab^2$
		
		\textbf{Solution:} GCF is $5ab^2$.
		
		$15a^2b^3 + 25ab^2 = 5ab^2(3ab) + 5ab^2(5) = 5ab^2(3ab + 5)$
		
		\textbf{Answer:} $5ab^2(3ab + 5)$

	\newpage

	\subsection*{Section 4: Factoring Trinomials \& Special Products}

		\textbf{Problem 8:} $x^2 + 11x + 24$
		
		\textbf{Solution:} Find two numbers that multiply to 24 and add to 11.
		
		Numbers: 3 and 8 (since $3 \times 8 = 24$ and $3 + 8 = 11$)
		
		\textbf{Answer:} $(x + 3)(x + 8)$

		\vspace{0.5cm}

		\textbf{Problem 9:} $2x^2 + 11x + 5$
		
		\textbf{Solution:} Use AC method. $a \times c = 2 \times 5 = 10$
		
		Find two numbers that multiply to 10 and add to 11: 10 and 1
		
		$2x^2 + 10x + 1x + 5$
		
		$2x(x + 5) + 1(x + 5)$
		
		\textbf{Answer:} $(2x + 1)(x + 5)$

	\newpage

	\subsection*{Section 5: Special Products}

		\textbf{Problem 10:} $x^2 + 16x + 64$
		
		\textbf{Solution:} Recognize as a perfect square trinomial.
		
		$x^2 + 2(x)(8) + 8^2 = (x + 8)^2$
		
		\textbf{Answer:} $(x + 8)^2$

		\vspace{0.5cm}

		\textbf{Problem 11:} $x^2 - 81$
		
		\textbf{Solution:} Recognize as difference of squares.
		
		$x^2 - 9^2 = (x + 9)(x - 9)$
		
		\textbf{Answer:} $(x + 9)(x - 9)$

	\newpage

	\subsection*{Section 6: Solving by Factoring}

		\textbf{Problem 12:} $x^2 + 9x + 20 = 0$
		
		\textbf{Solution:}
		
		Factor: $(x + 4)(x + 5) = 0$
		
		Set each factor equal to zero:
		
		$x + 4 = 0$ or $x + 5 = 0$
		
		$x = -4$ or $x = -5$
		
		\textbf{Answer:} $x = -4$ or $x = -5$

		\vspace{0.5cm}

		\textbf{Problem 13:} $2x^2 - 18 = 0$
		
		\textbf{Solution:}
		
		Factor out GCF: $2(x^2 - 9) = 0$
		
		Recognize difference of squares: $2(x + 3)(x - 3) = 0$
		
		Set each factor equal to zero:
		
		$x + 3 = 0$ or $x - 3 = 0$
		
		$x = -3$ or $x = 3$
		
		\textbf{Answer:} $x = -3$ or $x = 3$

	\subsection*{Section 7: Word Problems}

		\textbf{Problem 14:}
		
		A rectangular garden has a length that is twice its width.
		
		The area of the garden is 98 square feet.
		
		Find the dimensions of the garden.
		
		\textbf{Solution:}
		
		Let $w = $ width in feet
		
		Then length $= 2w$ feet
		
		Area equation: $w \cdot 2w = 98$
		
		$2w^2 = 98$
		
		$w^2 = 49$
		
		$w^2 - 49 = 0$
		
		Factor: $(w + 7)(w - 7) = 0$
		
		$w = -7$ or $w = 7$
		
		Width must be positive, so $w = 7$ feet
		
		Length $= 2(7) = 14$ feet
		
		\textbf{Answer:} Width = 7 feet, Length = 14 feet

		\vspace{0.5cm}


		\textbf{Problem 15:}
		
		A rocket is launched upward from the ground.
		
		Its height $h$ in feet after $t$ seconds is given by $h = -4t^2 + 32t$.
		
		At what time is the rocket at a height of 64 feet?
		
		\textbf{Solution:}
		
		Set $h = 64$: $64 = -4t^2 + 32t$
		
		Rewrite in standard form: $-4t^2 + 32t - 64 = 0$
		
		Factor out GCF of $-4$: $-4(t^2 - 8t + 16) = 0$
		
		Factor the trinomial (perfect square): $-4(t - 4)^2 = 0$
		
		Set factor equal to zero: $(t - 4)^2 = 0$
		
		$t - 4 = 0$
		
		$t = 4$
		
		\textbf{Answer:} The rocket is at 64 feet at $t = 4$ seconds
