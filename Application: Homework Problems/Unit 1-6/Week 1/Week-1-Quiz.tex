	    \subsection*{Properties of Real Numbers}
        	\begin{itemize}
            	\item Commutative Property: $a + b = b + a$; \quad $ab = ba$
                \item Associative Property: $(a + b) + c = a + (b + c)$; \quad $(ab)c = a(bc)$
                \item Distributive Property: $a(b + c) = ab + ac$
                \item Identity Property: $a + 0 = a$; \quad $a \cdot 1 = a$
                \item Inverse Property: $a + (-a) = 0$; \quad $a \cdot \frac{1}{a} = 1$ (for $a \neq 0$)
            \end{itemize}
	
	    \subsection*{Combining Like Terms}
        	Like terms have the same variable(s) raised to the same power(s).\\\\
            To combine like terms, add or subtract their coefficients.

    \section*{Solution Guide}

        \subsection*{Part A (1.1): Order of Operations (PEMDAS)}

            \textbf{1.} $9 + 3 \times 2$

                \textit{Solution:}\\
                Multiply first (M before A in PEMDAS):\\
                $9 + 3 \times 2 = 9 + 6 = 15$

                \fbox{Answer: 15}

                \vspace{0.5cm}

            \textbf{2.} $(6 + 8) \div 2$

                \textit{Solution:}\\
                Parentheses first, then divide:\\
                $(6 + 8) \div 2 = 14 \div 2 = 7$

                \fbox{Answer: 7}

                \vspace{0.5cm}

            \textbf{3.} $5^2 - (3 \times 2)$

                \textit{Solution:}\\
                Parentheses first, then exponents, then subtract:\\
                $5^2 - (3 \times 2) = 5^2 - 6 = 25 - 6 = 19$

                \fbox{Answer: 19}

                \vspace{0.5cm}

            \textbf{4.} $(18 \div 3) + (4 \times 2)$

                \textit{Solution:}\\
                Evaluate each parenthesis, then add:\\
                $(18 \div 3) + (4 \times 2) = 6 + 8 = 14$

                \fbox{Answer: 14}

        \subsection*{Part B (1.2): Properties of Real Numbers}

            \textbf{1.} $a + b = b + a$

                \fbox{Answer: Commutative Property of Addition}

                \vspace{0.5cm}

            \textbf{2.} $(2 + 5) + 7 = 2 + (5 + 7)$

                \fbox{Answer: Associative Property of Addition}

                \vspace{0.5cm}

            \textbf{3.} $6 \times 1 = 6$

                \fbox{Answer: Identity Property of Multiplication}

                \vspace{0.5cm}

            \textbf{4.} $3 \times (4 + 2) = 3 \times 4 + 3 \times 2$

                \fbox{Answer: Distributive Property}

                \vspace{0.5cm}

            \textbf{5.} $x + (-x) = 0$

                \fbox{Answer: Inverse Property of Addition}

        \subsection*{Part C (1.3): Combining Like Terms}

            \textbf{1.} $4x + 7x$

                \textit{Solution:}\\
                Both terms have the same variable $x$, so add the coefficients:\\
                $4x + 7x = (4 + 7)x = 11x$

                \fbox{Answer: $11x$}

                \vspace{0.5cm}

            \textbf{2.} $12a - 3a + 5$

                \textit{Solution:}\\
                Combine the $a$ terms (the 5 is a constant and stays separate):\\
                $12a - 3a + 5 = (12 - 3)a + 5 = 9a + 5$

                \fbox{Answer: $9a + 5$}

                \vspace{0.5cm}

            \textbf{3.} $5n + 2m - n + 3m$

                \textit{Solution:}\\
                Combine the $n$ terms and the $m$ terms separately:\\
                $5n + 2m - n + 3m = (5n - n) + (2m + 3m) = 4n + 5m$

                \fbox{Answer: $4n + 5m$}

                \vspace{0.5cm}

            \textbf{4.} $7x - 2x + 6 - 3$

                \textit{Solution:}\\
                Combine the $x$ terms and the constant terms separately:\\
                $7x - 2x + 6 - 3 = (7 - 2)x + (6 - 3) = 5x + 3$

                \fbox{Answer: $5x + 3$}

                \newpage

        \subsection*{Part D: Word Problems}

            \textbf{1.} A bookshelf's width is $3x - 2$ inches and the length is $5x + 4$ inches. Write a simplified expression for the perimeter in inches.

                \textit{Solution:}\\
                Perimeter of a rectangle: $P = 2(\text{width}) + 2(\text{length})$\\
                $P = 2(3x - 2) + 2(5x + 4)$\\
                $P = 6x - 4 + 10x + 8$\\
                $P = (6x + 10x) + (-4 + 8)$\\
                $P = 16x + 4$

                \fbox{Answer: $(16x + 4)$ inches}

                \vspace{0.5cm}

            \textbf{2.} A car rental company charges \$35 per day plus a one-time \$25 insurance fee. Write a simplified expression for the total cost of renting a car for $x$ days.

                \textit{Solution:}\\
                Total cost = (cost per day $\times$ number of days) + insurance fee\\
                Total cost $= 35x + 25$

                \fbox{Answer: $(35x + 25)$ dollars}

                \newpage

        \subsection*{Part E: Long Division Practice}

            \textbf{1.} Divide $648$ by $3$.

                \textit{Solution:}\\
                \begin{verbatim}
                      216
                    ------
                  3 | 648
                      6      (3 × 2 = 6)
                      --
                      04
                       3     (3 × 1 = 3)
                      --
                       18
                       18    (3 × 6 = 18)
                       --
                        0
                \end{verbatim}

                Step by step:\\
                • 6 ÷ 3 = 2 (write 2 above)\\
                • 4 ÷ 3 = 1 remainder 1 (write 1 above, bring down 8 to make 18)\\
                • 18 ÷ 3 = 6 (write 6 above)\\
                • Remainder = 0

                \fbox{Answer: 216}

                \newpage

        \subsection*{Bonus Challenge (Optional)}

            \textbf{1.} $3(2x + 5) - 4x + 6$

                \textit{Solution:}\\
                First, use the distributive property:\\
                $3(2x + 5) - 4x + 6$\\
                $= 6x + 15 - 4x + 6$\\
                \\
                Now combine like terms:\\
                $= (6x - 4x) + (15 + 6)$\\
                $= 2x + 21$

                \fbox{Answer: $2x + 21$}
