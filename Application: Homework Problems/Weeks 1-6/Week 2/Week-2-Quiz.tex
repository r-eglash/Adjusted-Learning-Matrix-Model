	\section*{Quick Reference: Solving Linear Equations}
	
        \subsection*{Steps for Solving Linear Equations}
            1. Simplify both sides (distribute, combine like terms)\\\\
            2. Move variable terms to one side\\\\
            3. Move constant terms to the other side\\\\
            4. Divide or multiply to isolate the variable\\\\
            5. Check your solution

	    \subsection*{Special Cases}
        	\begin{itemize}
            	\item \textbf{One Solution:} When you get $x = \text{number}$
                \item \textbf{No Solution:} When you get a false statement (e.g., $5 = -3$)
                \item \textbf{Infinitely Many Solutions:} When you get a true statement (e.g., $0 = 0$)
            \end{itemize}
	
	    \subsection*{Working with Fractions}
        	To eliminate fractions, multiply both sides by the LCD (Least Common Denominator).

    \section*{Solution Guide}

        \subsection*{Part A: Solve (variables on one side)}

            \textbf{1.} $4(x + 3) = 28$

                \textit{Solution:}\\
                Distribute the 4:\\
                $4x + 12 = 28$\\
                \\
                Subtract 12 from both sides:\\
                $4x = 16$\\
                \\
                Divide both sides by 4:\\
                $x = 4$

                \fbox{Answer: $x = 4$}

                \vspace{0.5cm}

            \textbf{2.} $0.25x - 7 = 6$

                \textit{Solution:}\\
                Add 7 to both sides:\\
                $0.25x = 13$\\
                \\
                Divide both sides by 0.25:\\
                $x = 52$

                \fbox{Answer: $x = 52$}

                \vspace{0.5cm}

            \textbf{3.} $\frac{x}{6} + 9 = 14$

                \textit{Solution:}\\
                Subtract 9 from both sides:\\
                $\frac{x}{6} = 5$\\
                \\
                Multiply both sides by 6:\\
                $x = 30$

                \fbox{Answer: $x = 30$}

                \newpage

        \subsection*{Part B: Solve (variables on both sides)}

            \textbf{4.} $7x - 5 = 2x + 30$

                \textit{Solution:}\\
                Subtract $2x$ from both sides:\\
                $5x - 5 = 30$\\
                \\
                Add 5 to both sides:\\
                $5x = 35$\\
                \\
                Divide both sides by 5:\\
                $x = 7$

                \fbox{Answer: $x = 7$}

                \vspace{0.5cm}

            \textbf{5.} $3(x - 2) = 2(x + 7)$

                \textit{Solution:}\\
                Distribute on both sides:\\
                $3x - 6 = 2x + 14$\\
                \\
                Subtract $2x$ from both sides:\\
                $x - 6 = 14$\\
                \\
                Add 6 to both sides:\\
                $x = 20$

                \fbox{Answer: $x = 20$}

                \newpage

            \textbf{6.} $5 - 2(x + 1) = x - 7$

                \textit{Solution:}\\
                Distribute the $-2$:\\
                $5 - 2x - 2 = x - 7$\\
                \\
                Combine like terms on the left:\\
                $3 - 2x = x - 7$\\
                \\
                Add $2x$ to both sides:\\
                $3 = 3x - 7$\\
                \\
                Add 7 to both sides:\\
                $10 = 3x$\\
                \\
                Divide both sides by 3:\\
                $x = \frac{10}{3}$

                \fbox{Answer: $x = \frac{10}{3}$ or $x = 3\frac{1}{3}$}

                \newpage

        \subsection*{Part C: Fractions \& Special Cases}

            \textbf{7.} $\frac{x}{3} + 2 = \frac{x}{6} + 7$

                \textit{Solution:}\\
                Multiply both sides by 6 (LCD of 3 and 6):\\
                $6 \cdot \frac{x}{3} + 6 \cdot 2 = 6 \cdot \frac{x}{6} + 6 \cdot 7$\\
                \\
                Simplify:\\
                $2x + 12 = x + 42$\\
                \\
                Subtract $x$ from both sides:\\
                $x + 12 = 42$\\
                \\
                Subtract 12 from both sides:\\
                $x = 30$

                \fbox{Answer: $x = 30$}

                \vspace{0.5cm}

            \textbf{8.} $\frac{3}{4}x - 5 = \frac{1}{2}x + 1$

                \textit{Solution:}\\
                Multiply both sides by 4 (LCD of 2 and 4):\\
                $4 \cdot \frac{3}{4}x - 4 \cdot 5 = 4 \cdot \frac{1}{2}x + 4 \cdot 1$\\
                \\
                Simplify:\\
                $3x - 20 = 2x + 4$\\
                \\
                Subtract $2x$ from both sides:\\
                $x - 20 = 4$\\
                \\
                Add 20 to both sides:\\
                $x = 24$

                \fbox{Answer: $x = 24$}

                \newpage

            \textbf{9.} $2(x - 4) + 6 = 2(x - 4) - 3$

                \textit{Solution:}\\
                Distribute on both sides:\\
                $2x - 8 + 6 = 2x - 8 - 3$\\
                \\
                Combine like terms:\\
                $2x - 2 = 2x - 11$\\
                \\
                Subtract $2x$ from both sides:\\
                $-2 = -11$\\
                \\
                This is a false statement, so there is no solution.

                \fbox{Answer: No solution}

                \newpage

        \subsection*{Part D: Word Problems}

            \textbf{10.} A streaming service costs \$12 per month, plus a one-time \$6 activation fee. A different streaming service costs \$15 per month, with no activation fee. After how many months $m$ will the total cost of both services be equal?

                \textit{Solution:}\\
                Set up the equation:\\
                First service: $12m + 6$\\
                Second service: $15m$\\
                \\
                When costs are equal:\\
                $12m + 6 = 15m$\\
                \\
                Subtract $12m$ from both sides:\\
                $6 = 3m$\\
                \\
                Divide both sides by 3:\\
                $m = 2$

                \fbox{Answer: $m = 2$ months}

                \vspace{0.5cm}

            \textbf{11.} A school fundraiser sells boxes of cookies for \$$x$ each. The club sold 18 boxes and raised a total of \$162. Find $x$, the price per box.

                \textit{Solution:}\\
                Set up the equation:\\
                (number of boxes) $\times$ (price per box) = total raised\\
                $18x = 162$\\
                \\
                Divide both sides by 18:\\
                $x = 9$

                \fbox{Answer: $x = \$9$ per box}
